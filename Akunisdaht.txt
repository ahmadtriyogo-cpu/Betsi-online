<script>
/* ---------------------- FIREBASE CONFIG ---------------------- */
const firebaseConfig = {
    apiKey: "PASTE_APIKEY_KAMU",
    authDomain: "PASTE_AUTHDOMAIN_KAMU",
    projectId: "PASTE_PROJECTID_KAMU",
};
firebase.initializeApp(firebaseConfig);
const db = firebase.firestore();
/* ------------------------------------------------------------- */

/* ---------------------- LOGIN SYSTEM ---------------------- */
const accounts = {
    "aht": "aht123",
    "isd": "isd123"
};

let currentRole = "";

function login() {
    const u = document.getElementById("username").value.trim();
    const p = document.getElementById("password").value.trim();

    if (accounts[u] && accounts[u] === p) {
        currentRole = u.toUpperCase();
        document.getElementById("loginCard").style.display = "none";
        document.getElementById("app").style.display = "block";

        document.getElementById("roleLabel").innerText = currentRole;

        if (currentRole === "AHT") {
            document.getElementById("ahtPanel").style.display = "block";
        } else {
            document.getElementById("ahtPanel").style.display = "none";
        }

        loadDrivers();

    } else {
        document.getElementById("loginMessage").innerText = "Username atau password salah!";
    }
}

function logout() {
    location.reload();
}

/* ---------------------- SAVE DRIVER ---------------------- */
async function saveDriver() {
    const name = document.getElementById("dName").value.trim();
    const phone = document.getElementById("dPhone").value.trim();
    const lambung = document.getElementById("dLambung").value.trim();
    const status = document.getElementById("dStatus").value;
    const note = document.getElementById("dNote").value.trim();

    if (!name || !phone) {
        alert("Nama dan HP wajib diisi");
        return;
    }

    await db.collection("drivers").add({
        name, phone, lambung, status, note,
        updatedAt: new Date(),
        updatedBy: currentRole
    });

    alert("Berhasil disimpan!");
    loadDrivers();
}

/* ---------------------- LOAD DRIVER LIST ---------------------- */
let driverCache = [];

function loadDrivers() {
    db.collection("drivers")
        .orderBy("updatedAt", "desc")
        .get()
        .then(snap => {
            driverCache = snap.docs.map(d => ({ id: d.id, ...d.data() }));
            renderList(driverCache);
        });
}

function renderList(list) {
    let html = "";

    list.forEach(d => {
        html += `
            <div class="card">
                <b>${d.name}</b> (${d.lambung || "-"})<br>
                HP: ${d.phone}<br>
                Status: ${d.status}<br>
                Note: ${d.note || "-"}<br>
                Update oleh: ${d.updatedBy}<br><br>

                <button class="green" onclick="updateTurun('${d.id}')">Turun</button>
            </div>
        `;
    });

    document.getElementById("driverList").innerHTML = html;
}

/* ---------------------- UPDATE TURUN (AHT & ISD) ---------------------- */
async function updateTurun(id) {
    const note = prompt("Keterangan turun:");

    await db.collection("drivers").doc(id).update({
        status: "Turun",
        note: note || "",
        updatedAt: new Date(),
        updatedBy: currentRole
    });

    loadDrivers();
}

/* ---------------------- SEARCH ---------------------- */
function filterList() {
    const key = document.getElementById("searchBox").value.toLowerCase();

    const filtered = driverCache.filter(d =>
        d.name.toLowerCase().includes(key) ||
        d.phone.includes(key) ||
        (d.lambung || "").toLowerCase().includes(key)
    );

    renderList(filtered);
}
